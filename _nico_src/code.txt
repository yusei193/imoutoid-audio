/**
 * imoutoid 動画（ニコニコ風弾幕プレイヤー）コメントバックエンド
 *  - GET  ?action=list&video_id=XXXXXXXXXXX   → {ok:true, comments:[...]}
 *  - POST {action:'post', video_id, time_sec, text, color, pos, size, uid}
 * スプレッドシート: imoutoid-nico-comments / シート: comments
 * 列: id | video_id | time_sec | text | color | pos | size | created_at | uid | status
 */

var SHEET_ID = '1cTr7esUYK8TPSQ-BNLQia3V4kXUJQJWMib83k1kDjo8';
var SHEET_NAME = 'comments';
var HEADER = ['id', 'video_id', 'time_sec', 'text', 'color', 'pos', 'size', 'created_at', 'uid', 'status'];
var MAX_LEN = 100;
var COLORS = ['white', 'red', 'pink', 'orange', 'yellow', 'green', 'cyan', 'blue', 'purple'];
var POSITIONS = ['naka', 'ue', 'shita'];
var SIZES = ['big', 'medium', 'small'];

function doGet(e) {
  return route(e, {});
}

function doPost(e) {
  var body = {};
  try {
    body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
  } catch (err) {
    body = {};
  }
  return route(e, body);
}

function route(e, body) {
  var p = (e && e.parameter) || {};
  // GET のクエリと POST の JSON の両方を受け付ける
  var data = {};
  var k;
  for (k in p) { if (Object.prototype.hasOwnProperty.call(p, k)) data[k] = p[k]; }
  for (k in body) { if (Object.prototype.hasOwnProperty.call(body, k)) data[k] = body[k]; }
  var action = data.action || 'list';
  var out;
  try {
    if (action === 'list') {
      out = { ok: true, server_time: nowStr(), comments: listComments(data.video_id) };
    } else if (action === 'post') {
      out = postComment(data);
    } else {
      out = { ok: false, error: 'unknown action' };
    }
  } catch (err) {
    out = { ok: false, error: String((err && err.message) || err) };
  }
  return respond(out, data.callback);
}

function respond(obj, callback) {
  var s = JSON.stringify(obj);
  if (callback && /^[A-Za-z_$][A-Za-z0-9_$]{0,40}$/.test(callback)) {
    return ContentService.createTextOutput(callback + '(' + s + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON);
}

function nowStr() {
  return Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM-dd HH:mm:ss');
}

function getSheet() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(HEADER);
  }
  return sh;
}

function validVideoId(v) {
  return typeof v === 'string' && /^[\w-]{11}$/.test(v);
}

function listComments(videoId) {
  if (!validVideoId(videoId)) return [];
  var cache = CacheService.getScriptCache();
  var key = 'c_' + videoId;
  var hit = cache.get(key);
  if (hit) {
    try { return JSON.parse(hit); } catch (e) {}
  }
  var sh = getSheet();
  var last = sh.getLastRow();
  var out = [];
  if (last >= 2) {
    var vals = sh.getRange(2, 1, last - 1, 10).getValues();
    for (var i = 0; i < vals.length; i++) {
      var r = vals[i];
      if (String(r[1]) !== videoId) continue;
      if (String(r[9]).toLowerCase() === 'hidden') continue;
      out.push({
        id: r[0],
        time_sec: Number(r[2]) || 0,
        text: String(r[3]),
        color: String(r[4] || 'white'),
        pos: String(r[5] || 'naka'),
        size: String(r[6] || 'medium')
      });
    }
  }
  out.sort(function (a, b) { return a.time_sec - b.time_sec; });
  try {
    var payload = JSON.stringify(out);
    if (payload.length < 95000) cache.put(key, payload, 120);
  } catch (e) {}
  return out;
}

function cleanText(raw) {
  var t = String(raw == null ? '' : raw);
  t = t.replace(/[\u0000-\u001F\u007F\u200B-\u200F\u2028\u2029]/g, ' ');
  t = t.replace(/[<>]/g, '');
  t = t.replace(/\s+/g, ' ').replace(/^\s+|\s+$/g, '');
  return t.slice(0, MAX_LEN);
}

function postComment(body) {
  var videoId = String(body.video_id || '');
  if (!validVideoId(videoId)) return { ok: false, error: '動画IDが不正です' };

  var text = cleanText(body.text);
  if (!text) return { ok: false, error: 'コメントが空です' };
  if (String(body.text || '').length > MAX_LEN * 2) return { ok: false, error: 'コメントが長すぎます' };

  var color = COLORS.indexOf(String(body.color)) >= 0 ? String(body.color) : 'white';
  var pos = POSITIONS.indexOf(String(body.pos)) >= 0 ? String(body.pos) : 'naka';
  var size = SIZES.indexOf(String(body.size)) >= 0 ? String(body.size) : 'medium';

  var t = Number(body.time_sec);
  if (!isFinite(t) || t < 0) t = 0;
  if (t > 86400) t = 86400;
  t = Math.round(t * 100) / 100;

  var uid = String(body.uid || '').slice(0, 40);
  if (!uid) uid = 'anon';

  var cache = CacheService.getScriptCache();
  var rateKey = 'r_' + uid;
  var lastAt = Number(cache.get(rateKey) || 0);
  var now = new Date().getTime();
  if (lastAt && now - lastAt < 10000) {
    return { ok: false, error: '投稿が続いています。10秒ほど待ってください' };
  }
  var hourKey = 'h_' + uid;
  var hourCount = Number(cache.get(hourKey) || 0);
  if (hourCount >= 80) return { ok: false, error: '1時間の投稿数の上限に達しました' };

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = getSheet();
    var last = sh.getLastRow();
    var id = 1;
    if (last >= 2) {
      var ids = sh.getRange(2, 1, last - 1, 1).getValues();
      for (var i = 0; i < ids.length; i++) {
        var n = Number(ids[i][0]);
        if (isFinite(n) && n >= id) id = n + 1;
      }
    }
    sh.appendRow([id, videoId, t, text, color, pos, size, nowStr(), uid, 'visible']);
    cache.remove('c_' + videoId);
    cache.put(rateKey, String(now), 60);
    cache.put(hourKey, String(hourCount + 1), 3600);
    return { ok: true, id: id };
  } finally {
    lock.releaseLock();
  }
}

/* ---- 管理用（エディタから手動実行）---- */
function setupSheet() {
  var sh = getSheet();
  var first = sh.getRange(1, 1, 1, HEADER.length).getValues()[0];
  var need = false;
  for (var i = 0; i < HEADER.length; i++) if (String(first[i]) !== HEADER[i]) need = true;
  if (need || sh.getLastRow() === 0) {
    sh.getRange(1, 1, 1, HEADER.length).setValues([HEADER]);
    sh.setFrozenRows(1);
    sh.getRange(1, 1, 1, HEADER.length).setFontWeight('bold');
  }
  return 'ok';
}

function countAll() {
  var sh = getSheet();
  return Math.max(0, sh.getLastRow() - 1);
}
