function doGet(e){var p=e.parameter;var s=PropertiesService.getScriptProperties();var d=Utilities.formatDate(new Date(),"Asia/Tokyo","yyyy-MM-dd");var v=String(p.vid||"");var k="v"+v;var n=Number(s.getProperty("hits")||1);var hit=false;if(p.key=="iia2026"&&p.set){n=Number(p.set);s.setProperty("hits",String(n));}else if(p.mode!="get"&&v&&s.getProperty(k)!=d){n=n+1;s.setProperty("hits",String(n));s.setProperty(k,d);hit=true;}return ContentService.createTextOutput(JSON.stringify({value:n,hit:hit})).setMimeType(ContentService.MimeType.JSON);}
